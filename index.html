// One pool, any prefab — no allocations mid-gameplay.
public class Pool<T> where T : Component
{
    private readonly Queue<T> _idle = new();
    private readonly T _prefab;

    public Pool(T prefab, int warm)
    {
        _prefab = prefab;
        for (int i = 0; i < warm; i++) _idle.Enqueue(Create());
    }

    public T Get(Vector3 at)
    {
        T item = _idle.Count > 0 ? _idle.Dequeue() : Create();
        item.transform.position = at;
        item.gameObject.SetActive(true);
        return item;
    }

    public void Release(T item)
    {
        item.gameObject.SetActive(false);
        _idle.Enqueue(item);
    }
}
